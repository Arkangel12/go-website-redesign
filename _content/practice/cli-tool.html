CLI Tool

A command-line utility with subcommands, flags, and formatted output — using only the standard library.

What you'll build

A note command with three subcommands:

  • note add <text> — append a note to a file
  • note list — print all notes numbered
  • note clear — wipe all notes (with a confirmation prompt)

main.go

package main

import (
	"bufio"
	"flag"
	"fmt"
	"os"
	"strings"
	"time"
)

const dataFile = "notes.txt"

func main() {
	if len(os.Args) < 2 {
		usage()
		os.Exit(1)
	}

	switch os.Args[1] {
	case "add":
		cmdAdd(os.Args[2:])
	case "list":
		cmdList()
	case "clear":
		cmdClear()
	default:
		fmt.Fprintf(os.Stderr, "unknown command: %s\n", os.Args[1])
		usage()
		os.Exit(1)
	}
}

func usage() {
	fmt.Fprintln(os.Stderr, "usage: note <command> [args]")
	fmt.Fprintln(os.Stderr, "commands: add, list, clear")
}

func cmdAdd(args []string) {
	fs := flag.NewFlagSet("add", flag.ExitOnError)
	fs.Parse(args)

	text := strings.Join(fs.Args(), " ")
	if text == "" {
		fmt.Fprintln(os.Stderr, "add: provide note text")
		os.Exit(1)
	}

	f, err := os.OpenFile(dataFile, os.O_APPEND|os.O_CREATE|os.O_WRONLY, 0644)
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	defer f.Close()

	line := fmt.Sprintf("[%s] %s\n", time.Now().Format("2006-01-02 15:04"), text)
	f.WriteString(line)
	fmt.Print("saved: ", line)
}

func cmdList() {
	f, err := os.Open(dataFile)
	if os.IsNotExist(err) {
		fmt.Println("no notes yet — try: note add <text>")
		return
	}
	if err != nil {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	defer f.Close()

	scanner := bufio.NewScanner(f)
	n := 0
	for scanner.Scan() {
		n++
		fmt.Printf("%3d  %s\n", n, scanner.Text())
	}
	if n == 0 {
		fmt.Println("no notes")
	}
}

func cmdClear() {
	fmt.Print("delete all notes? [y/N] ")
	var answer string
	fmt.Scanln(&answer)
	if strings.ToLower(answer) != "y" {
		fmt.Println("cancelled")
		return
	}
	if err := os.Remove(dataFile); err != nil && !os.IsNotExist(err) {
		fmt.Fprintln(os.Stderr, err)
		os.Exit(1)
	}
	fmt.Println("cleared")
}

Try it

go run main.go add Buy groceries
go run main.go add "Read Effective Go"
go run main.go list
go run main.go clear

Key concepts

Subcommand dispatch
A switch on os.Args[1] is the simplest way to route subcommands. Each handler gets its own flag.FlagSet so flags don't collide.
os.OpenFile flags
O_APPEND|O_CREATE|O_WRONLY opens the file for appending, creating it if absent. No need to read the whole file into memory.
bufio.Scanner
Reads a file line-by-line without loading it all at once — the right default for text files of unknown size.
